// Force HTTPS for WordPress Admin
define('FORCE_SSL_ADMIN', true);

// Force WordPress to use HTTPS URLs
define('WP_HOME', 'https://henieskinlab.com');
define('WP_SITEURL', 'https://henieskinlab.com');

// Fix for sites behind reverse proxy/load balancer (Cloudflare, etc.)
if (isset($_SERVER['HTTP_X_FORWARDED_PROTO']) && $_SERVER['HTTP_X_FORWARDED_PROTO'] === 'https') {
    $_SERVER['HTTPS'] = 'on';
}

// Ensure HTTPS is detected correctly
if (isset($_SERVER['HTTP_X_FORWARDED_SSL']) && $_SERVER['HTTP_X_FORWARDED_SSL'] === 'on') {
    $_SERVER['HTTPS'] = 'on';
}
